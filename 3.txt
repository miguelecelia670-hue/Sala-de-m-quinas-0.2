  const message=prompt('Mensagem ao comprador (opcional):')||null;

  const existing=await sb.from('sm_counteroffers').select('id').eq('proposal_id',proposal.id).is('buyer_response',null).limit(1);

}

let paymentInProgress = new Set();
async function confirmSalePayment(sale, actionButton){
  if(!(staff.role==='owner'||(staff.role==='employee'&&staff.can_manage_sales===true))){
    notify('Você não possui permissão para gerenciar vendas.');return;
  }
  if(sale.status!=='negotiating'||sale.stock_deducted||sale.paid_at){
    notify('Esta venda não pode ser confirmada neste estado.');return;
  }
  if(paymentInProgress.has(sale.id))return;
  if(!window.confirm(`Confirmar pagamento de ${money(sale.total_amount)}?\n\nVenda: ${sale.id}\nQuantidade: ${sale.quantity}\n\nEsta ação registrará o pagamento e descontará o estoque. Não pode ser desfeita pelo painel.`))return;
  paymentInProgress.add(sale.id);
  actionButton.disabled=true;
  actionButton.textContent='Processando…';
  try{
    const {error}=await sb.rpc('sm_confirm_sale_payment',{p_sale_id:sale.id});
    if(error)throw error;
    await render();
    notify('Pagamento confirmado. Estoque e auditoria atualizados pelo banco.');
  }catch(e){
    notify('Pagamento não confirmado: '+(e.message||'Erro desconhecido'));
    actionButton.disabled=false;
    actionButton.textContent='Confirmar pagamento';
  }finally{paymentInProgress.delete(sale.id)}
}
